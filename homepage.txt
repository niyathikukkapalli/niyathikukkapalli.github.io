<script>
  const text   = "Hi! I'm Niyathi!";
  let i        = 0;
  const typed  = document.getElementById('bio-typed');
  const cursor = document.getElementById('bio-cursor');

  function type() {
    if (i < text.length) {
      typed.textContent += text[i];
      i++;
      setTimeout(type, 80);
    } else {
      setTimeout(() => { cursor.style.display = 'none'; }, 1500);
    }
  }
  setTimeout(type, 300);
</script>
