<script>
  function openLb(title, cat, desc, imgSrc) {
    document.getElementById('lb-title').textContent = title;
    document.getElementById('lb-cat').textContent   = cat;
    document.getElementById('lb-desc').textContent  = desc;
    document.getElementById('lb-img').src           = imgSrc;
    document.getElementById('lb-img').alt           = title;
    document.getElementById('lb').classList.add('open');
  }

  function closeLb(e) {
    if (e.target === document.getElementById('lb')) {
      document.getElementById('lb').classList.remove('open');
    }
  }
</script>
